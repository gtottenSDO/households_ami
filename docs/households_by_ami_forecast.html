Households by AMI Forecast

Purpose

This script is to generate a forecast of households by AMI based on the most recent three year average at the state level.

Show code 
source("_common.R")

Pull Data from Postgres

The necessary tables are stored in the SDO Postgres database.

con <- sdodemog::connect_to_sdo_db()

hh_by_ten_ami <- pull_db_table(

  filter_model_id = TRUE
)

hh_projections <- pull_db_table(
  schema = "estimates",
  table = "household_projections"
)

xwalk_hna_region <- pull_db_table(
  "xwalk",

  mutate(region = str_sub(region_code, -1))

First, check that the state totals match the sum of the regions for cost burdened households and total households.

Create AMI Allocation Tables

The forecast methodology used is to allocate SDO household projections by the most recent three year average of households by AMI group within tenure from the PUMS data. This was done for simplicity as initially forecasts were run econometrically to determine percentages over time. These tended to be quite stable with relatively little trend in the percentages over time, so the three year average was used instead to simplify the process and reduce potential noise.

The general process is to begin by calculating by year the percentage of all households by tenure, and percentage of households by AMI group within each tenure group, then calculate the most recent three year averages, and then create an allocation table that can be used to allocate SDO household projections1. These are generated with a function that outputs the multiple dataframes into a list. These generate the following tables into the list, which have the following structure:

  1. hh_ami_pct: historical percentages of households by tenure and AMI group within tenure by year and region. This is used to allocate down households, first by allocating total households into tenure groups, then by tenure groups into AMI groups within each tenure.
  2. hh_ami_pct_fcst: the three year average of the most recent percentages in hh_ami_pct to be used for forecasting future years.
  3. hh_ami_allocation_tbl: a combined table of historical percentages from hh_ami_pct and forecasted percentages from hh_ami_pct_fcst for all years in the household projections. This is used to allocate the households from the SDO household projections into tenure and AMI groups based on the calculated percentages determined from the PUMS data.
Show code 
calc_ami_allocation <- function(df) {
  hh_ami_pct <- df |>
    filter(!(is.na(ami_group))) |>
    select(year, tenure, ami_group, region, hh_ami_ten = total_hh) |>
    mutate(hh_tot = sum(hh_ami_ten), .by = c(year, region)) |>
    mutate(
      hh_ten = sum(hh_ami_ten),
      .by = c(year, region, tenure)
    ) |>
    mutate(
      pct_hh_ten = hh_ten / hh_tot,
      pct_hh_ami_ten = hh_ami_ten / hh_ten
    )

  hh_ami_pct_fcst <- hh_ami_pct |>
    filter(year >= max(year) - 2) |>
    summarize(
      hh_ami_ten_avg = mean(hh_ami_ten),
      hh_tot_avg = mean(hh_tot),
      hh_ten_avg = mean(hh_ten),
      .by = c(region, tenure, ami_group)
    ) |>
    mutate(
      pct_hh_ten = hh_ten_avg / hh_tot_avg,
      pct_hh_ami_ten = hh_ami_ten_avg / hh_ten_avg
    )

  hh_ami_allocation_tbl <- bind_rows(
    hh_ami_pct |>
      select(year, region, tenure, ami_group, pct_hh_ten, pct_hh_ami_ten),
    expand_grid(
      year = c((max(hh_ami_pct$year) + 1):max(hh_projections$year)),
      hh_ami_pct_fcst |>
        select(region, tenure, ami_group, pct_hh_ten, pct_hh_ami_ten)
    )
  )

  return(
    list(
      "hh_ami_pct" = hh_ami_pct,
      "hh_ami_pct_fcst" = hh_ami_pct_fcst,
      "hh_ami_allocation_tbl" = hh_ami_allocation_tbl
    )
  )
}

hh_ami_tbls <- hh_by_ten_ami |>
  calc_ami_allocation()

Allocate Household Projections

Next the household projections from SDO are allocated into AMI groups using the allocation tables created above. This is done first at the region level, then summed to the state level, and finally combined into a single table. This is done by first allocating the region, then summing the regions into the state level totals.

Region Level Allocation

This is done by filtering the household projections to only total households (household_type_id == 0 and age_group_id == 0), joining to the HNA region crosswalk to get the region for each county, summarizing to get total households by year and region, then joining to the allocation table to get the percentages, and finally calculating the allocated households by multiplying total households by the percentage of households by tenure and percentage of households by AMI group within each tenure.

Show code 
hh_ami_forecast_region <- hh_projections |>
  filter(
    area_code != 0,
    household_type_id == 0,
    age_group_id == 0
  ) |>
  left_join(
    xwalk_hna_region |>
      select(
        area_code = countyfips,
        region_code
      ) |>
      mutate(region = str_sub(region_code, -1)),
    by = "area_code"
  ) |>
  summarize(
    total_households = sum(total_households, na.rm = TRUE),
    .by = c(year, region)
  ) |>
  left_join(
    hh_ami_tbls$hh_ami_allocation_tbl,
    by = c("year", "region")
  ) |>
  mutate(
    hh_allocated = total_households * pct_hh_ten * pct_hh_ami_ten
  )

State Level Allocation

Next, the region level allocations are summed to the state level by summing across all regions for each year, tenure, and AMI group. Then the total households by year are calculated, followed by the percentages of households by tenure and by AMI group within each tenure.

Show code 
hh_ami_forecast_state <- hh_ami_forecast_region |>
  summarize(
    region = "0",
    hh_allocated = sum(hh_allocated),
    .by = c(year, tenure, ami_group)
  ) |>
  mutate(
    total_households = sum(hh_allocated),
    .by = year
  ) |>
  mutate(
    pct_hh_ten = sum(hh_allocated) / total_households,
    pct_hh_ami_ten = hh_allocated / sum(hh_allocated),
    .by = c(year, tenure)
  )

hh_ami_forecast <- bind_rows(
  hh_ami_forecast_state,
  hh_ami_forecast_region
) |>
  arrange(year, region, tenure, ami_group)

Quality Assurance Checks

Several QA checks are performed to ensure that the allocations are correct.

Show code 
# check allocations = total
hh_ami_forecast |>
  summarize(
    hh_allocated = sum(hh_allocated),
    .by = c(year, region, total_households)
  ) |>
  mutate(check = total_households - hh_allocated) |>
  filter(abs(check) > 1)

# check against existing SDO table
hh_ami_existing <- pull_db_table(
  schema = "econ",
  table = "hh_ami_forecast",
  filter_model_id = TRUE
)

# Join and compare tables
hh_ami_compare <- hh_ami_forecast |>
  left_join(
    hh_ami_existing |>
      select(
        year,
        region,
        tenure,
        ami_group,
        hh_allocated_existing = hh_allocated
      ),
    by = c("year", "region", "tenure", "ami_group")
  ) |>
  mutate(
    diff = hh_allocated - hh_allocated_existing
  ) |>
  filter(abs(diff) > 1)

Write Data to Postgres and CSV

Finally, the resulting household AMI forecast table is written to the SDO Postgres database and also saved as a CSV file.

Show code 
sdodemog::sdo_db_write_table(
  connect_to_sdo_db(),
  schema = "econ",
  table = "hh_ami_forecast",
  df = hh_ami_forecast,
  vintage = 2024,
  latest = TRUE
)


write_csv(
  hh_ami_forecast,
  "data/hh_ami_forecast.csv"
)
